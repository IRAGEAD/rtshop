===== server/world/serverconfig/fantasy-server.toml
#The format of this file. The mod keeps it up to date, carrying your settings over when an update renames or moves one. Leave it as it is.
#Range: > 1
configVersion = 1

#Quests: the narrator (in the shipped quests Professor Thornquill, the wandering bestiarist) writes to players in their Field Journal when they meet or tame a mount, with tasks about it. Rewards and hints are scaled in fantasy-balance.toml; the quests and the people in them are data (data/<namespace>/fantasy_quest and fantasy_quest_speaker).
[quests]
	#Quests and their narrator. Off: no introduction, no letters, no quest tracker, and no quest NPCs or foes. Players' quest progress is kept for when quests are turned back on.
	enabled = false
	#Greet each player with the narrator's introduction (and its first quest) the first time they join the world.
	introduction = true
	#Let quests place their people and foes in the world: the narrator's camp, a captive to rescue, a named foe. Off: such an objective counts once the player stands where it would have been.
	encounters = true
	#Keep a mount's Riders Encyclopedia entry to what anyone who met one could tell (its picture, name, description, temperament, size and riders, and the player's own field notes) until the player finishes one of the mount's quests: where it lives, how to tame and breed it, how it rides, its abilities and stats open then. How to tame one opens early once a quest asks the player to tame one. Mounts without quests, or switched off, show in full, and nothing is held back while quests are off or encyclopedia.revealAll is on.
	lockEncyclopedia = true

	#Earn mounts through their quests (off unless enabled is true). While on, a player who hasn't finished one of a kind of mount's quests can still tame, saddle, equip and whistle for one, but can't do what is switched on below until they finish one. Mounts with no quests are never held back, and nothing is while quests.enabled is false. A quest that asks the player to ride one or raise one to a level opens that early, so a quest is never held back by this.
	[quests.earnMounts]
		#Make players finish one of a kind of mount's quests before they may do what is switched on below with mounts of that kind. Off: riding, breeding and bond items are never held back, whatever the switches below say.
		enabled = false
		#Only while enabled is true: hold back riding. The player can't take the reins of a tamed mount of the kind (its driver's seat; a passenger seat on someone else's mount stays open). Checked as they get on. Taming one by riding it still works.
		riding = true
		#Only while enabled is true: hold back breeding. The player can't feed mounts of the kind into love mode.
		breeding = true
		#Only while enabled is true: hold back bond items. The player can't give bond items (Bond Shards and Crystals) to mounts of the kind; they still level up as they follow and fight.
		bondItems = true

#Who may ride and drive mounts.
[riding]
	#Who may take a mount's driver seat: OWNER, TR

#Every skill of every mount. Each dial in this file multiplies what a skill adds (its stat raises and ability bonuses): 1.0 = as shipped, 0.5 = half as strong, 2.0 = twice, 0 = nothing. 	#Every skill of every mount.
	#Range: 0.0 ~ 10.0

	#Range: 0.0 ~ 10.0

	#Gentle Soul: +5% Health per rank (3 ranks, 1 point each)
	#Range: 0.0 ~ 10.0

	#Soothing Touch: Regrowth cooldown -12% per rank (2 ranks, 1 point each, from level 10)
	#Range: 0.0 ~ 10.0
	soothing_touch = 1.0
	#Font of Life: Regrowth cooldown -20%, reshaped (not scaled), +8% Health (4 points, from level 22)
